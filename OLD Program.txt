function fetchSettingsAndPlaylist() {
    fetch('/settings', { cache: 'no-store' })
    .then(response => response.json())
    .then(data => {
        document.getElementById('audioToggle').checked = data.audioEnabled;
        document.getElementById('clockDelay').value = data.clockDelay;

        patternConfigs.chase_1_10       = { speed: data.c1Speed, qty: data.c1Qty, size: data.c1Size, attack: 0.15, release: 0.01, music: data.c1Music, freq: data.c1Freq, multiplier: data.c1Mult };
        patternConfigs.chase_10_1       = { speed: data.c10Speed, qty: data.c10Qty, size: data.c10Size, attack: 0.15, release: 0.01, music: data.c10Music, freq: data.c10Freq, multiplier: data.c10Mult };
        patternConfigs.center_out_chase = { speed: data.ccoSpeed, qty: data.ccoQty, size: data.ccoSize, attack: 0.15, release: 0.01, music: data.ccoMusic, freq: data.ccoFreq, multiplier: data.ccoMult };
        patternConfigs.vu_meter         = { speed: data.vuSpeed, qty: 1, size: 1, attack: data.vuAttack, release: data.vuRelease, music: data.vuMusic, freq: data.vuFreq, multiplier: 1 };
        patternConfigs.vu_meter_rev     = { speed: data.vurSpeed, qty: 1, size: 1, attack: data.vurAttack, release: data.vurRelease, music: data.vurMusic, freq: data.vurFreq, multiplier: 1 };
        patternConfigs.vu_dual          = { speed: data.vudSpeed, qty: 1, size: 1, attack: data.vudAttack, release: data.vudRelease, music: data.vudMusic, freq: data.vudFreq, multiplier: 1 };
        patternConfigs.vu_center_out    = { speed: data.vucoSpeed, qty: 1, size: 1, attack: data.vucoAttack, release: data.vucoRelease, music: data.vucoMusic, freq: data.vucoFreq, multiplier: 1 };

        const radios = document.getElementsByName('patternRadio');

            if (r.value === currentActivePattern) r.checked = true;
        }

        if (data.playlist) {
            playlistItems = data.playlist;
        }
        if (data.playlistPlaying !== undefined) {
            isPlaying = data.playlistPlaying;

        }

function switchPage(page) {
    if (page === 'playlist') {
        document.getElementById('pageMain').style.display = 'none';
        document.getElementById('pagePlaylist').style.display = 'block';
    } else {
        if (isPlaying) {
            togglePlaylistPlay(); 
        }
        document.getElementById('pagePlaylist').style.display = 'none';
        document.getElementById('pageMain').style.display = 'block';
    }
}

function handleAudioToggle() {
    const isAudioEnabled = document.getElementById('audioToggle').checked;
    const speedSlider = document.getElementById('clockDelay');
    speedSlider.disabled = isAudioEnabled;

    const freqInput = document.getElementById('sharedFreq');
    const freqContainer = document.getElementById('freqContainer');
    freqInput.disabled = !isAudioEnabled;
    freqContainer.style.opacity = isAudioEnabled ? '1.0' : '0.3';

    const audioOnlyPatterns = ['center_out_chase', 'vu_meter', 'vu_meter_rev', 'vu_dual', 'vu_center_out'];
    const radios = document.getElementsByName('patternRadio');
    for (const r of radios) {
        if (audioOnlyPatterns.includes(r.value)) {
            r.disabled = !isAudioEnabled;
            const parentLabel = r.closest('.radio-label');
            if (parentLabel) {
                if (!isAudioEnabled) {
                    parentLabel.classList.add('disabled');
                } else {
                    parentLabel.classList.remove('disabled');
                }
            }
        }
    }

    if (!isAudioEnabled && audioOnlyPatterns.includes(currentActivePattern)) {
        document.querySelector('input[name="patternRadio"][value="moving_dot"]').checked = true;
        onPatternChange();
    }
}

function getSelectedPattern() {
    const radios = document.getElementsByName('patternRadio');
    for (const r of radios) {
        if (r.checked) return r.value;
    }
    return 'moving_dot';
}

void handleUpdate() {
  if (server.hasArg("plain")) {
    String payload = server.arg("plain");
    JsonDocument doc;
    DeserializationError error = deserializeJson(doc, payload);
    
    if (!error) {
      audioEnabled = doc["audioEnabled"].as<bool>();
      clockDelay = doc["clockDelay"].as<float>();
      antifreeze = constrain(doc["antifreeze"].as<int>(), 1, 9);
      activePattern = doc["pattern"].as<String>();
      
      bool newPlaylistPlaying = doc["playlistPlaying"].as<bool>();
      if (newPlaylistPlaying && !playlistPlaying) {
        // Started playing playlist from stopped state
        playlistTrackStartTime = millis();
        int firstSel = -1;
        for (int i = 0; i < playlistCount; i++) {
          if (playlist[i].selected) { firstSel = i; break; }
        }
        if (firstSel != -1) {
          currentPlaylistIndex = firstSel;
          loadPlaylistTrack(currentPlaylistIndex);
        }
      }
      playlistPlaying = newPlaylistPlaying;
      playlistRandom = doc["playlistRandom"].as<bool>();
      playlistDuration = constrain(doc["playlistDuration"].as<int>(), 10, 60);

      if (doc.containsKey("playlist")) {
        JsonArray plArray = doc["playlist"].as<JsonArray>();
        playlistCount = 0;
        for (JsonObject item : plArray) {
          if (playlistCount < MAX_PLAYLIST_ITEMS) {
            strlcpy(playlist[playlistCount].name, item["name"] | "Track", sizeof(playlist[playlistCount].name));
            playlist[playlistCount].selected = item["selected"] | true;
            strlcpy(playlist[playlistCount].pattern, item["pattern"] | "moving_dot", sizeof(playlist[playlistCount].pattern));
            playlist[playlistCount].audioEnabled = item["audioEnabled"] | true;
            playlist[playlistCount].clockDelay = item["clockDelay"] | 1.0;
            playlist[playlistCount].antifreeze = item["antifreeze"] | 2;
            playlist[playlistCount].speed = item["speed"] | 0.1;
            playlist[playlistCount].qty = item["qty"] | 1;
            playlist[playlistCount].size = item["size"] | 1;
            strlcpy(playlist[playlistCount].music, item["music"] | "BASS", sizeof(playlist[playlistCount].music));
            playlist[playlistCount].freq = item["freq"] | 50;
            playlist[playlistCount].mult = item["mult"] | 2;
            playlist[playlistCount].attack = item["attack"] | 0.15;
            playlist[playlistCount].release = item["release"] | 0.01;
            playlistCount++;
          }
        }
      }
      
      mdSpeed = doc["mdSpeed"].as<float>(); mdQty = doc["mdQty"].as<int>(); mdSize = doc["mdSize"].as<int>(); mdMusic = doc["mdMusic"].as<String>(); mdFreq = doc["mdFreq"].as<int>(); mdMult = constrain(doc["mdMult"].as<int>(), 1, 9);
      c1Speed = doc["c1Speed"].as<float>(); c1Qty = doc["c1Qty"].as<int>(); c1Size = doc["c1Size"].as<int>(); c1Music = doc["c1Music"].as<String>(); c1Freq = doc["c1Freq"].as<int>(); c1Mult = constrain(doc["c1Mult"].as<int>(), 1, 9);
      c10Speed = doc["c10Speed"].as<float>(); c10Qty = doc["c10Qty"].as<int>(); c10Size = doc["c10Size"].as<int>(); c10Music = doc["c10Music"].as<String>(); c10Freq = doc["c10Freq"].as<int>(); c10Mult = constrain(doc["c10Mult"].as<int>(), 1, 9);
      ccoSpeed = doc["ccoSpeed"].as<float>(); ccoQty = doc["ccoQty"].as<int>(); ccoSize = doc["ccoSize"].as<int>(); ccoMusic = doc["ccoMusic"].as<String>(); ccoFreq = doc["ccoFreq"].as<int>(); ccoMult = constrain(doc["ccoMult"].as<int>(), 1, 9);
      
      vuSpeed = doc["vuSpeed"].as<float>(); vuAttack = doc["vuAttack"].as<float>(); vuRelease = doc["vuRelease"].as<float>(); vuMusic = doc["vuMusic"].as<String>(); vuFreq = doc["vuFreq"].as<int>();
      vurSpeed = doc["vurSpeed"].as<float>(); vurAttack = doc["vurAttack"].as<float>(); vurRelease = doc["vurRelease"].as<float>(); vurMusic = doc["vurMusic"].as<String>(); vurFreq = doc["vurFreq"].as<int>();
      vudSpeed = doc["vudSpeed"].as<float>(); vudAttack = doc["vudAttack"].as<float>(); vudRelease = doc["vudRelease"].as<float>(); vudMusic = doc["vudMusic"].as<String>(); vudFreq = doc["vudFreq"].as<int>();
      vucoSpeed = doc["vucoSpeed"].as<float>(); vucoAttack = doc["vucoAttack"].as<float>(); vucoRelease = doc["vucoRelease"].as<float>(); vucoMusic = doc["vucoMusic"].as<String>(); vucoFreq = doc["vucoFreq"].as<int>();

      // Automatically persist settings and playlist changes to EEPROM
      preferences.begin("lightshow", false);
      preferences.putBool("audioEnabled", audioEnabled);
      preferences.putFloat("clockDelay", clockDelay);
      preferences.putInt("antifreeze", antifreeze);
      preferences.putString("pattern", activePattern);

      preferences.end();

      savePlaylistToEEPROM();

      server.send(200, "text/plain", "Updated successfully");
    } else {
      server.send(400, "text/plain", "Invalid JSON");
    }
  } else {
    server.send(400, "text/plain", "Bad Request");
  }
}

void loadPlaylistTrack(int index) {
  if (index < 0 || index >= playlistCount) return;
  PlaylistItem &item = playlist[index];
  
  activePattern = String(item.pattern);
  audioEnabled = item.audioEnabled;
  clockDelay = item.clockDelay;
  antifreeze = item.antifreeze;

  if (activePattern == "moving_dot") {
    mdSpeed = item.speed; mdQty = item.qty; mdSize = item.size; mdMusic = item.music; mdFreq = item.freq; mdMult = item.mult;
  } else if (activePattern == "chase_1_10") {
    c1Speed = item.speed; c1Qty = item.qty; c1Size = item.size; c1Music = item.music; c1Freq = item.freq; c1Mult = item.mult;
  } else if (activePattern == "chase_10_1") {
    c10Speed = item.speed; c10Qty = item.qty; c10Size = item.size; c10Music = item.music; c10Freq = item.freq; c10Mult = item.mult;
  } else if (activePattern == "center_out_chase") {
    ccoSpeed = item.speed; ccoQty = item.qty; ccoSize = item.size; ccoMusic = item.music; ccoFreq = item.freq; ccoMult = item.mult;
  } else if (activePattern == "vu_meter") {
    vuSpeed = item.speed; vuAttack = item.attack; vuRelease = item.release; vuMusic = item.music; vuFreq = item.freq;
  } else if (activePattern == "vu_meter_rev") {
    vurSpeed = item.speed; vurAttack = item.attack; vurRelease = item.release; vurMusic = item.music; vurFreq = item.freq;
  } else if (activePattern == "vu_dual") {
    vudSpeed = item.speed; vudAttack = item.attack; vudRelease = item.release; vudMusic = item.music; vudFreq = item.freq;
  } else if (activePattern == "vu_center_out") {
    vucoSpeed = item.speed; vucoAttack = item.attack; vucoRelease = item.release; vucoMusic = item.music; vucoFreq = item.freq;
  }
}

void advancePlaylistTrack() {
  int selectedIndices[MAX_PLAYLIST_ITEMS];
  int selCount = 0;
  for (int i = 0; i < playlistCount; i++) {
    if (playlist[i].selected) {
      selectedIndices[selCount++] = i;
    }
  }

  if (selCount == 0) {
    playlistPlaying = false;
    return;
  }

  if (playlistRandom) {
    int rIdx = random(0, selCount);
    currentPlaylistIndex = selectedIndices[rIdx];
  } else {
    int nextIdx = -1;
    for (int i = 0; i < selCount; i++) {
      if (selectedIndices[i] > currentPlaylistIndex) {
        nextIdx = selectedIndices[i];
        break;
      }
    }
    if (nextIdx == -1) {
      nextIdx = selectedIndices[0];
    }
    currentPlaylistIndex = nextIdx;
  }

  loadPlaylistTrack(currentPlaylistIndex);
  playlistTrackStartTime = millis();
}

  playlistCount = preferences.getInt("plCount", 0);

  for (int i = 0; i < playlistCount && i < MAX_PLAYLIST_ITEMS; i++) {

void clearAllChannels() {
  for (int i = 0; i < 10; i++) {
    digitalWrite(channelPins[i], LOW);
  }
}

bool checkAudioBeat() {
  unsigned long currentMillis = millis();

  if (currentMillis - lastSampleTime >= 1) {
    lastSampleTime = currentMillis;
    rawSampleVal = analogRead(AUDIO_PIN);

    if (currentMusicMode == "FULL") {
      filteredSample = rawSampleVal; 
    } else {
      float cutoff = max(5, freqThreshold);
      float rc = 1.0 / (2.0 * 3.14159265 * cutoff);
      float dt = 0.001;
      float alpha = dt / (rc + dt);
      filteredSample = filteredSample + alpha * (rawSampleVal - filteredSample);
    }

    int sampleVal = (currentMusicMode == "FULL") ? rawSampleVal : (int)filteredSample;
    if (sampleVal > signalMax) signalMax = sampleVal;
    if (sampleVal < signalMin) signalMin = sampleVal;
  }

  if (currentMillis - audioWindowStart >= 10) {
    currentPeakToPeak = signalMax - signalMin; 
    signalMax = 0;
    signalMin = 4095;
    audioWindowStart = currentMillis;

    if (generatedSubBeatsCount > 0 && multiplier > 1) {
      if (currentMillis >= nextSubBeatTargetTime) {
        generatedSubBeatsCount--;
        nextSubBeatTargetTime = currentMillis + (establishedInterval / multiplier);
        return true;
      }
    }

    if (currentPeakToPeak > 120) {
      if (currentMillis - lastAcceptedBeatTime >= 200) {
        if (lastRealBeatTime > 0) {
          unsigned long interval = currentMillis - lastRealBeatTime;
          if (interval >= 150 && interval <= 3000) {
            beatIntervalHistory[historyIndex] = interval;
            historyIndex = (historyIndex + 1) % HISTORY_SIZE;
            if (historyCount < HISTORY_SIZE) historyCount++;
          }
        }
        lastRealBeatTime = currentMillis;
        lastAcceptedBeatTime = currentMillis;

        if (historyCount > 0) {
          unsigned long sum = 0;
          for (int i = 0; i < historyCount; i++) {
            sum += beatIntervalHistory[i];
          }
          establishedInterval = sum / historyCount;
        } else {
          establishedInterval = 500;
        }

        if (multiplier > 1 && establishedInterval > 200) {
          generatedSubBeatsCount = multiplier - 1;
          nextSubBeatTargetTime = currentMillis + (establishedInterval / multiplier);
        }

        return true;
      }
    }
  }
  return false;
}

void executePatternStep(bool beatHeard) {
  clearAllChannels();

  if (activePattern == "moving_dot") {
    int effQty = constrain(mdQty, 1, 10);
    int dotSize = constrain(mdSize, 1, 5);
    if (beatHeard) {
      dotDirection = -dotDirection;
    }

    for (int q = 0; q < effQty; q++) {
      int baseIdx = (currentStep + (q * 10) / effQty) % 10;
      for (int s = 0; s < dotSize; s++) {
        int idx = (baseIdx + s) % 10;
        digitalWrite(channelPins[idx], HIGH);
      }
    }
    
    currentStep = (currentStep + dotDirection + 10) % 10;
  }
  else if (activePattern == "chase_1_10") {
    int effQty = constrain(c1Qty, 1, 10);
    int dotSize = constrain(c1Size, 1, 5);

    for (int q = 0; q < effQty; q++) {
      int baseIdx = (currentStep + (q * 10) / effQty) % 10;
      for (int s = 0; s < dotSize; s++) {
        int idx = (baseIdx + s) % 10;
        digitalWrite(channelPins[idx], HIGH);
      }
    }
    currentStep = (currentStep + 1) % 10;
  }
  else if (activePattern == "chase_10_1") {
    int effQty = constrain(c10Qty, 1, 10);
    int dotSize = constrain(c10Size, 1, 5);

    for (int q = 0; q < effQty; q++) {
      int baseIdx = (9 - currentStep - (q * 10) / effQty + 10) % 10;
      for (int s = 0; s < dotSize; s++) {
        int idx = (baseIdx - s + 10) % 10;
        digitalWrite(channelPins[idx], HIGH);
      }
    }
    currentStep = (currentStep + 1) % 10;
  }
  else if (activePattern == "center_out_chase") {
    int effQty = constrain(ccoQty, 1, 10);
    int dotSize = constrain(ccoSize, 1, 5);
    if (beatHeard) {
      currentStep = 0; // Restart burst from center on beat
    }
    
    if (currentStep < 5) {
      for (int q = 0; q < effQty; q++) {
        int stepOffset = currentStep - (q * 2);
        if (stepOffset >= 0 && stepOffset < 5) {
          int lCenter = 4 - stepOffset;
          int rCenter = 5 + stepOffset;
          
          for (int s = 0; s < dotSize; s++) {
            int lIdx = lCenter - s;
            int rIdx = rCenter + s;
            if (lIdx >= 0) digitalWrite(channelPins[lIdx], HIGH);
            if (rIdx < 10) digitalWrite(channelPins[rIdx], HIGH);
          }
        }
      }
    }
  }
  else if (activePattern.startsWith("vu_")) {
    float targetLevel = map(constrain(currentPeakToPeak, 0, 180), 0, 180, 0, 10);
    
    if (targetLevel > masterVuSmooth) {
      masterVuSmooth += VU_ATTACK_RATE * (targetLevel - masterVuSmooth); 
    } else {
      masterVuSmooth += VU_RELEASE_RATE * (targetLevel - masterVuSmooth); 
    }
    if (masterVuSmooth < 0.3) masterVuSmooth = 0;
    int smoothLevel = constrain((int)(masterVuSmooth + 0.5), 0, 10);

    unsigned long currentMillis = millis();

    if (activePattern == "vu_meter") {
      int level = smoothLevel;
      if (level > vuPeakLevel) {
        vuPeakLevel = level;
        lastVuPeakFallTime = currentMillis;
      } else if (currentMillis - lastVuPeakFallTime > 150) { 
        if (vuPeakLevel > 0) vuPeakLevel--;
        lastVuPeakFallTime = currentMillis;
      }
      for (int i = 0; i < level; i++) {
        digitalWrite(channelPins[i], HIGH);
      }
      if (vuPeakLevel > level && vuPeakLevel <= 10 && vuPeakLevel > 0) {
        digitalWrite(channelPins[vuPeakLevel - 1], HIGH);
      }
    }
    else if (activePattern == "vu_meter_rev") {
      int level = smoothLevel;
      if (level > vurPeakLevel) {
        vurPeakLevel = level;
        lastVurPeakFallTime = currentMillis;
      } else if (currentMillis - lastVurPeakFallTime > 150) {
        if (vurPeakLevel > 0) vurPeakLevel--;
        lastVurPeakFallTime = currentMillis;
      }
      for (int i = 0; i < level; i++) {
        digitalWrite(channelPins[9 - i], HIGH);
      }
      if (vurPeakLevel > level && vurPeakLevel <= 10 && vurPeakLevel > 0) {
        digitalWrite(channelPins[9 - (vurPeakLevel - 1)], HIGH);
      }
    }
    else if (activePattern == "vu_dual") {
      int level = smoothLevel / 2; 
      level = constrain(level, 0, 5);
      for (int i = 0; i < level; i++) {
        digitalWrite(channelPins[i], HIGH);
        digitalWrite(channelPins[9 - i], HIGH);
      }
    }
    else if (activePattern == "vu_center_out") {
      int level = smoothLevel / 2; 
      level = constrain(level, 0, 5);
      for (int i = 0; i < level; i++) {
        digitalWrite(channelPins[4 - i], HIGH); 
        digitalWrite(channelPins[5 + i], HIGH); 
      }
    }
  }
}

void run240VDisplayPattern() {
  unsigned long currentMillis = millis();

  // Handle Playlist Timing & Transition
  if (playlistPlaying) {
    if (currentMillis - playlistTrackStartTime >= (unsigned long)(playlistDuration * 1000)) {
      advancePlaylistTrack();
    }
  }

  float currentSpeed = 0.1;

  if (activePattern == "moving_dot") {
    currentSpeed = mdSpeed; freqThreshold = mdFreq; multiplier = mdMult; currentMusicMode = mdMusic;
  } else if (activePattern == "chase_1_10") {
    currentSpeed = c1Speed; freqThreshold = c1Freq; multiplier = c1Mult; currentMusicMode = c1Music;
  } else if (activePattern == "chase_10_1") {
    currentSpeed = c10Speed; freqThreshold = c10Freq; multiplier = c10Mult; currentMusicMode = c10Music;
  } else if (activePattern == "center_out_chase") {
    currentSpeed = ccoSpeed; freqThreshold = ccoFreq; multiplier = ccoMult; currentMusicMode = ccoMusic;
  } else if (activePattern == "vu_meter") {
    currentSpeed = vuSpeed; freqThreshold = vuFreq; multiplier = 1; currentMusicMode = vuMusic;
    VU_ATTACK_RATE = vuAttack; VU_RELEASE_RATE = vuRelease;
  } else if (activePattern == "vu_meter_rev") {
    currentSpeed = vurSpeed; freqThreshold = vurFreq; multiplier = 1; currentMusicMode = vurMusic;
    VU_ATTACK_RATE = vurAttack; VU_RELEASE_RATE = vurRelease;
  } else if (activePattern == "vu_dual") {
    currentSpeed = vudSpeed; freqThreshold = vudFreq; multiplier = 1; currentMusicMode = vudMusic;
    VU_ATTACK_RATE = vudAttack; VU_RELEASE_RATE = vudRelease;
  } else if (activePattern == "vu_center_out") {
    currentSpeed = vucoSpeed; freqThreshold = vucoFreq; multiplier = 1; currentMusicMode = vucoMusic;
    VU_ATTACK_RATE = vucoAttack; VU_RELEASE_RATE = vucoRelease;
  }

  if (!audioEnabled) {
    if (currentMillis - lastStepTime >= (unsigned long)(clockDelay * 1000)) {
      lastStepTime = currentMillis;
      executePatternStep(false);
    }
  } 
  else {
    bool beat = checkAudioBeat();

    if (activePattern.startsWith("vu_")) {
      executePatternStep(false); 
    } 
    else {
      if (beat) {
        lastBeatTime = currentMillis;
        executePatternStep(true);
      } 
      else {
        if (activePattern == "moving_dot") {
          if (currentMillis - lastStepTime >= (unsigned long)(currentSpeed * 1000)) {
            lastStepTime = currentMillis;
            executePatternStep(false); 
          }
        }
        else if (activePattern == "center_out_chase") {
          if (currentMillis - lastStepTime >= (unsigned long)(currentSpeed * 1000)) {
            lastStepTime = currentMillis;
            if (currentStep < 5) {
              currentStep++;
              executePatternStep(false);
            }
          }
        }

        if (currentMillis - lastBeatTime >= (unsigned long)(antifreeze * 1000)) {
          lastBeatTime = currentMillis; 
          executePatternStep(false); 
        }
      }
    }
  }
}

void WiFiEvent(WiFiEvent_t event) {
    switch (event) {
        case ARDUINO_EVENT_WIFI_AP_STACONNECTED:
            Serial.println("Client connected to LightShow AP!");
            break;
        case ARDUINO_EVENT_WIFI_AP_STADISCONNECTED:
            Serial.println("Client disconnected from AP.");
            break;
        default:
            break;
    }
}

void setup() {
  Serial.begin(115200);
  delay(500);

  esp_task_wdt_config_t wdt_config = {
      .timeout_ms = 5000,
      .idle_core_mask = (1 << 0),
      .trigger_panic = true
  };
  esp_task_wdt_init(&wdt_config);
  esp_task_wdt_add(NULL);

  analogSetAttenuation(ADC_11db);
  pinMode(AUDIO_PIN, INPUT);

  for (int i = 0; i < 10; i++) {
    pinMode(channelPins[i], OUTPUT);
    digitalWrite(channelPins[i], LOW);
  }

  preferences.begin("lightshow", true);
  audioEnabled = preferences.getBool("audioEnabled", true);
  clockDelay = preferences.getFloat("clockDelay", 1.0);
  antifreeze = preferences.getInt("antifreeze", 2);
  activePattern = preferences.getString("pattern", "moving_dot");
  
  mdSpeed = preferences.getFloat("mdSpeed", 0.1); mdQty = preferences.getInt("mdQty", 1); mdSize = preferences.getInt("mdSize", 1); mdMusic = preferences.getString("mdMusic", "BASS"); mdFreq = preferences.getInt("mdFreq", 50); mdMult = preferences.getInt("mdMult", 2);
  c1Speed = preferences.getFloat("c1Speed", 0.1); c1Qty = preferences.getInt("c1Qty", 1); c1Size = preferences.getInt("c1Size", 1); c1Music = preferences.getString("c1Music", "BASS"); c1Freq = preferences.getInt("c1Freq", 50); c1Mult = preferences.getInt("c1Mult", 2);
  c10Speed = preferences.getFloat("c10Speed", 0.1); c10Qty = preferences.getInt("c10Qty", 1); c10Size = preferences.getInt("c10Size", 1); c10Music = preferences.getString("c10Music", "BASS"); c10Freq = preferences.getInt("c10Freq", 50); c10Mult = preferences.getInt("c10Mult", 2);
  ccoSpeed = preferences.getFloat("ccoSpeed", 0.1); ccoQty = preferences.getInt("ccoQty", 1); ccoSize = preferences.getInt("ccoSize", 1); ccoMusic = preferences.getString("ccoMusic", "BASS"); ccoFreq = preferences.getInt("ccoFreq", 50); ccoMult = preferences.getInt("ccoMult", 2);
  
  vuSpeed = preferences.getFloat("vuSpeed", 0.1); vuAttack = preferences.getFloat("vuAttack", 0.15); vuRelease = preferences.getFloat("vuRelease", 0.01); vuMusic = preferences.getString("vuMusic", "BASS"); vuFreq = preferences.getInt("vuFreq", 50);
  vurSpeed = preferences.getFloat("vurSpeed", 0.1); vurAttack = preferences.getFloat("vurAttack", 0.15); vurRelease = preferences.getFloat("vurRelease", 0.01); vurMusic = preferences.getString("vurMusic", "BASS"); vurFreq = preferences.getInt("vurFreq", 50);
  vudSpeed = preferences.getFloat("vudSpeed", 0.1); vudAttack = preferences.getFloat("vudAttack", 0.15); vudRelease = preferences.getFloat("vudRelease", 0.01); vudMusic = preferences.getString("vudMusic", "BASS"); vudFreq = preferences.getInt("vudFreq", 50);
  vucoSpeed = preferences.getFloat("vucoSpeed", 0.1); vucoAttack = preferences.getFloat("vucoAttack", 0.15); vucoRelease = preferences.getFloat("vucoRelease", 0.01); vucoMusic = preferences.getString("vucoMusic", "BASS"); vucoFreq = preferences.getInt("vucoFreq", 50);
  preferences.end();

  loadPlaylistFromEEPROM();

  // Default to playing the playlist on power up if present, otherwise start on Page 1
  if (playlistCount > 0) {
    playlistPlaying = true;
    playlistTrackStartTime = millis();
    int firstSel = -1;
    for (int i = 0; i < playlistCount; i++) {
      if (playlist[i].selected) { firstSel = i; break; }
    }
    if (firstSel != -1) {
      currentPlaylistIndex = firstSel;
      loadPlaylistTrack(currentPlaylistIndex);
    }
  } else {
    playlistPlaying = false;
  }

  WiFi.onEvent(WiFiEvent);
  WiFi.mode(WIFI_AP);
  WiFi.softAP(apSSID, apPassword);
  WiFi.setTxPower(WIFI_POWER_19_5dBm);
  
  Serial.println("");
  Serial.print("Access Point IP address: ");
  Serial.println(WiFi.softAPIP());

  server.on("/", HTTP_GET, handleRoot);
  server.on("/ping", HTTP_GET, handlePing);
  server.on("/settings", HTTP_GET, handleGetSettings);
  server.on("/update", HTTP_POST, handleUpdate);
  server.on("/save", HTTP_POST, handleSave);

  server.begin();
  Serial.println("HTTP server started successfully!");
}

void loop() {
  esp_task_wdt_reset(); 
  server.handleClient();
  run240VDisplayPattern();
  yield();
}
